
import { useEffect, useState } from "react";
import { useLoaderData } from "react-router";
import { getInstalledAppIdsFromLS, getSortByDownloadsLS, setSortByDownloadsLS } from "../../utilities/utilities";
import Title from "../../components/Title/Title";
import InstalledApps from "../InstalledApps/InstalledApps";
// import { toast } from "react-toastify";

const Installation = () => {
    const apps = useLoaderData()

    const [installedIds, setInstalledIds] = useState(() => getInstalledAppIdsFromLS().map(i => parseInt(i)))
    const [installedApps, setInstalledApps] = useState([])
    const sortApps = (apps, sortType) => {
        if (sortType === "high-to-low") {
            return [...apps].sort((a, b) => b.downloads - a.downloads)
        }
        else if (sortType === "low-to-high") {
            return [...apps].sort((a, b) => a.downloads - b.downloads)
        }
        else {
            return apps
        }
    }

    useEffect(() => {

        const updateInstallApps = apps.filter(app => installedIds.includes(app.id))
        const sortType = getSortByDownloadsLS()
        setInstalledApps(sortApps(updateInstallApps, sortType))

        // if (sortType === "high-to-low") {
        //     setInstalledApps(sortApps(updateInstallApps, "high-to-low"))
        // }
        // else if (sortType === "low-to-high") {
        //     setInstalledApps(sortApps(updateInstallApps, "low-to-high"))
        // }
        // else {
        //     setInstalledApps(sortApps(updateInstallApps))
        // }
        
   }, [apps, installedIds])
    // console.log(installedApps)

    const handleUninstall = id => {
        const updateInstallIds = installedIds.filter(installedId => installedId !== id)
        localStorage.setItem("appIds", JSON.stringify(updateInstallIds))
        setInstalledIds(updateInstallIds)
    }

    const handleSortByDownload = sortByDownload => {
        const updateInstallApps = apps.filter(app => installedIds.includes(app.id))
        setInstalledApps(sortApps(updateInstallApps, sortByDownload))
        setSortByDownloadsLS(sortByDownload)
        /* if (sortByDownload === "high-to-low") {
            setInstalledApps(sortApps(updateInstallApps, "high-to-low"))
            setSortByDownloadsLS("high-to-low")
        }
        else if (sortByDownload === "low-to-high") {
            setInstalledApps(sortApps(updateInstallApps, "low-to-high"))
            setSortByDownloadsLS("low-to-high")
        }
        else {
            setInstalledApps(sortApps(updateInstallApps))
            setSortByDownloadsLS("default")
        } */
    
    }

    return (
        <>
            <div className="bg-base-200 py-15">
                {
                    installedApps.length > 0
                        ?
                        <Title title="Your Installed Apps" subTitle="Explore All Trending Apps on the Market developed by us"></Title>
                        :
                        <div className="text-center">
                            <h1 className="text-5xl font-bold mb-4">You have not installed any app</h1>
                        </div>
                }
                <div className="py-12 max-w-400 mx-auto">
                    <div className="flex justify-between items-center py-5">
                        <h2 className="font-bold text-2xl">({installedApps.length}) Installed Apps </h2>
                        {
                            installedApps.length > 0
                            &&
                            <select defaultValue="Medium" className="select select-md">
                                <option disabled={true}>Sort By Download</option>
                                <option onClick={() => handleSortByDownload("default")}>Default</option>
                                <option onClick={() => handleSortByDownload("high-to-low")}>High to Low</option>
                                <option onClick={() => handleSortByDownload("low-to-high")}>Low to High</option>
                            </select>
                        }

                    </div>
                    <div className="flex flex-col gap-y-4">
                        {
                            installedApps.map(installedApp => <InstalledApps key={installedApp.id} installedApp={installedApp} handleUninstall={handleUninstall} ></InstalledApps>)
                        }
                    </div>
                </div>

            </div>
        </>

    );
};

export default Installation;
